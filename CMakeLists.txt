cmake_minimum_required(VERSION 3.24)
project(SLXSynthForge VERSION 0.4.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Gui Quick QuickControls2 Concurrent Svg OpenGL)

set(SLX_CORE_BINARY "${CMAKE_CURRENT_SOURCE_DIR}/sdk/windows-mingw/SiliconLogiX.SynthForge.Core.dll.a" CACHE FILEPATH "Approved core import library")
set(SLX_CORE_DLL "${CMAKE_CURRENT_SOURCE_DIR}/sdk/windows-mingw/SiliconLogiX.SynthForge.Core.dll" CACHE FILEPATH "Matching approved core runtime")
if(NOT EXISTS "${SLX_CORE_BINARY}" OR NOT EXISTS "${SLX_CORE_DLL}")
    message(FATAL_ERROR "Set SLX_CORE_BINARY and SLX_CORE_DLL to a matching approved SDK pair")
endif()
if(SLX_CORE_BINARY STREQUAL "${CMAKE_CURRENT_SOURCE_DIR}/sdk/windows-mingw/SiliconLogiX.SynthForge.Core.dll.a")
    if(NOT WIN32 OR NOT MINGW OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR
       NOT CMAKE_CXX_COMPILER_VERSION VERSION_EQUAL "13.1.0" OR NOT Qt6_VERSION VERSION_EQUAL "6.12.0")
        message(FATAL_ERROR "The included SDK requires Windows x64, Qt 6.12.0 and MinGW GCC 13.1.0")
    endif()
endif()
add_library(slx-synth-core INTERFACE)
target_include_directories(slx-synth-core INTERFACE core/include)
target_link_libraries(slx-synth-core INTERFACE "${SLX_CORE_BINARY}" Qt6::Core Qt6::Gui)

qt_add_executable(slx-synthforge src/main.cpp src/controller.cpp src/workflow.cpp src/editor.cpp
    src/experiments.cpp src/inspectorworkflow.cpp src/assemblyview.cpp src/assemblyview.h src/controller.h resources.qrc)
target_link_libraries(slx-synthforge PRIVATE slx-synth-core Qt6::Quick Qt6::QuickControls2 Qt6::Concurrent Qt6::Svg Qt6::OpenGL)
target_compile_definitions(slx-synthforge PRIVATE SLX_APP_VERSION="${PROJECT_VERSION}")
set_target_properties(slx-synthforge PROPERTIES WIN32_EXECUTABLE TRUE MACOSX_BUNDLE TRUE)
if(WIN32)
    add_custom_command(TARGET slx-synthforge POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "${SLX_CORE_DLL}" "$<TARGET_FILE_DIR:slx-synthforge>")
    qt_add_executable(slx-icon-maker EXCLUDE_FROM_ALL scripts/make-icon.cpp)
    target_link_libraries(slx-icon-maker PRIVATE Qt6::Gui Qt6::Svg)
    target_sources(slx-synthforge PRIVATE assets/brand/appicon.rc)
endif()
if(MSVC)
    target_compile_options(slx-synthforge PRIVATE /W4 /utf-8)
else()
    target_compile_options(slx-synthforge PRIVATE -Wall -Wextra -Wpedantic "-ffile-prefix-map=${CMAKE_CURRENT_SOURCE_DIR}=.")
endif()

include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    qt_add_executable(core-tests tests/core_tests.cpp)
    target_link_libraries(core-tests PRIVATE slx-synth-core Qt6::Test)
    target_compile_definitions(core-tests PRIVATE SLX_ASSET_DIR="${CMAKE_CURRENT_SOURCE_DIR}/SLX_SynthForge_PowerBoard")
    add_test(NAME core-contract COMMAND core-tests -o core-contract.txt,txt)
    qt_add_executable(assembly-tests tests/assembly_tests.cpp)
    target_link_libraries(assembly-tests PRIVATE slx-synth-core Qt6::Test)
    target_compile_definitions(assembly-tests PRIVATE SLX_ASSET_DIR="${CMAKE_CURRENT_SOURCE_DIR}/SLX_SynthForge_PowerBoard" SLX_PROJECT_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
    add_test(NAME editable-board-inspection COMMAND assembly-tests -o editable-board-inspection.txt,txt)
    add_test(NAME app-startup COMMAND slx-synthforge --smoke-test --assets "${CMAKE_CURRENT_SOURCE_DIR}/SLX_SynthForge_PowerBoard")
    set_tests_properties(app-startup PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    find_package(Python3 COMPONENTS Interpreter QUIET)
    if(Python3_FOUND)
        execute_process(COMMAND "${Python3_EXECUTABLE}" -c "import numpy" RESULT_VARIABLE _numpy_found OUTPUT_QUIET ERROR_QUIET)
        if(_numpy_found EQUAL 0)
            qt_add_executable(workflow-tests tests/workflow_tests.cpp src/controller.cpp src/workflow.cpp src/editor.cpp src/experiments.cpp src/inspectorworkflow.cpp src/controller.h)
            target_include_directories(workflow-tests PRIVATE src)
            target_link_libraries(workflow-tests PRIVATE slx-synth-core Qt6::Test Qt6::Concurrent)
            target_compile_definitions(workflow-tests PRIVATE SLX_ASSET_DIR="${CMAKE_CURRENT_SOURCE_DIR}/SLX_SynthForge_PowerBoard" SLX_PROJECT_DIR="${CMAKE_CURRENT_SOURCE_DIR}" SLX_PYTHON_EXECUTABLE="${Python3_EXECUTABLE}")
            add_test(NAME local-ml-workflow COMMAND workflow-tests -o local-ml-workflow.txt,txt)
        endif()
    endif()
endif()
